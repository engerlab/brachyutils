126    def load_from_ctegsphant(self, pth_file:str):
127        r"""
128        Purpose: 
129            to load a file with extension .egsphant into a BrachyEgsphant object
130        Input:
131            - pth_file := directory path to the .egsphant file
132        """
133        assert os.path.splitext(pth_file)[-1] == ".egsphant", "target file does not have .egsphant extension"
134        
135        with open(pth_file, "r") as egsphant:
136            # first line describes how many materials are used
137            self.num_materials = int(egsphant.readline().strip())
138            
139            # load each material line by line
140            for i in range(self.num_materials):
141                self.material_dict[egsphant.readline().strip()] = i
142            
143            egsphant.readline()
144            
145            # load number of voxels
146            self.num_voxels = np.array([int(i) for i in egsphant.readline().strip().split()])
147            
148            # load the axis grid points
149            self._sanity_axis = np.array([
150                np.array([float(x) for x in egsphant.readline().strip().split()], dtype=np.float32),
151                np.array([float(y) for y in egsphant.readline().strip().split()], dtype=np.float32),
152                np.array([float(z) for z in egsphant.readline().strip().split()], dtype=np.float32)
153                ], dtype=object)
154            self._sanity_axis = np.flip(self._sanity_axis, axis=0)
155            
156            self.topleft = np.array(
157                [
158                    self._sanity_axis[2][0],
159                    self._sanity_axis[1][0],
160                    self._sanity_axis[0][0]
161                ], dtype=np.float32)
162            
163            self.voxel_size = np.array(
164                [
165                    self._sanity_axis[2][1] - self._sanity_axis[2][0],
166                    self._sanity_axis[1][1] - self._sanity_axis[1][0],
167                    self._sanity_axis[0][1] - self._sanity_axis[0][0]
168                ]
169            )
170            # this line maybe useless in the future
171            self.axis = self.calculateAxis()
172            # {for debugging
173            # print(f"The axis calculated from calculateAxis() are \n {self.axis}")
174            # print(f"The axis from the text file are: \n {self._sanity_axis}")
175            # print(f"the size of the axis in the z, y, x for axis from calcAxis() are {self.axis[0].shape}, {self.axis[1].shape}, {self.axis[2].shape}")
176            # print(f"the size of the axis in the z, y, x for axis from file are {self._sanity_axis[0].shape}, {self._sanity_axis[1].shape}, {self._sanity_axis[2].shape}")
177            # }
178            assert np.isclose(np.concatenate(self.axis), np.concatenate(self._sanity_axis), rtol=1e-1).all(), "axis is not the same"
179 
180            # prepare empty matricies to hold material and density images
181            self.material_matrix = npzeros((self.num_voxels[2], self.num_voxels[1], self.num_voxels[0]), dtype=int)
182            self.density_matrix = npzeros((self.num_voxels[2], self.num_voxels[1], self.num_voxels[0]), dtype=np.float32)
183
184            # load the material composition data in to the matrix
185            for k in range(self.num_voxels[2]):
186                for j in range(self.num_voxels[1]):
187                    self.material_matrix[k][j] = list(egsphant.readline().strip())
188                egsphant.readline()
189                
190            # load the density data into the matrix
191            for k in range(self.num_voxels[2]):
192                for j in range(self.num_voxels[1]):
193                    self.density_matrix[k][j] = egsphant.readline().strip().split()
194                egsphant.readline()

205    def calculateAxis(self):
206        r"""
207        Purpose: will calculate the axies coordinates for a BrachyEgsphant object.
208        Input: 
209            - self := it should have the following keys and values:
210                {"grid":,
211                "topleft":,
212                "voxel_size":}
213        Output: 
214            - axes:numpy.array() := 
215            [[z_min:voxel_size:z_max],
216            [y_min:voxel_size:y_max],
217            [x_min:voxel_size:x_max]] 
218        """
219        # calculate the end point of axis in 3D space
220        axes_end = np.array(
221            self.topleft +  self.num_voxels* self.voxel_size + self.voxel_size # one voxel size is added because np.arange stops at an index before the end  
222        )
223        axes = np.empty(len(axes_end), dtype=object)
224        for i in range(len(axes_end)):
225            axes[i] = np.arange(self.topleft[len(axes_end)-1-i], axes_end[len(axes_end)-1-i], self.voxel_size[len(axes_end)-1-i], dtype=np.float32)
226        
227        return axes

322    def crop_by_index(self, index_range:np.array, inplace:Optional[bool]=True):
323        r"""
324        Purpose: 
325            given a range of indicies (mix and max on each axis), this function will crop
326            material and density matricies and will adjust the rest of the attributes accordingly. 
327        Inputs:
328            - self: BrachyEgsphant object
329            - index_range := a 3 x 2 array holding the min and max on x, y and axis
330                [[x_min, x_max], [y_min, y_max], [z_min, z_max]] 
331        Output:
332            - Void := will crop out the material and density maps of self to have the range of the index range. 
333                it will also update the num_voxels, topleft and axis. only voxel_size will not change
334        Dependencies:
335            - None
336        """
337        new_origin_index = index_range[:, 0].astype(int)
338        assert np.all(new_origin_index >= 0), "new origin index cannot be negative, please report this bug"
339
340        new_ending_index = index_range[:, 1].astype(int)
341        assert np.all(new_ending_index >= 0), "new ending index cannot be negative, please report this bug"
342
343        # update the attributes 
344        if inplace:
345            self.material_matrix = self.material_matrix[
346                new_origin_index[2]:new_ending_index[2], # z
347                new_origin_index[1]:new_ending_index[1], # y
348                new_origin_index[0]:new_ending_index[0] # x
349            ]          
350            self.density_matrix = self.density_matrix[
351                new_origin_index[2]:new_ending_index[2], # z
352                new_origin_index[1]:new_ending_index[1], # y
353                new_origin_index[0]:new_ending_index[0] # x
354            ]    
355            self.topleft = np.array([
356                self.axis[2][new_origin_index[0]], # x
357                self.axis[1][new_origin_index[1]], # y
358                self.axis[0][new_origin_index[2]] # z
359            ])
360            self.num_voxels = np.flip(self.material_matrix.shape, 0)    
361            self.axis = self.calculateAxis()      
362        else:
363            new_obj = BrachyEgsphant()
364            new_obj.material_matrix = self.material_matrix[
365                new_origin_index[2]:new_ending_index[2],
366                new_origin_index[1]:new_ending_index[1],
367                new_origin_index[0]:new_ending_index[0]
368            ]          
369            new_obj.density_matrix = self.density_matrix[
370                new_origin_index[2]:new_ending_index[2],
371                new_origin_index[1]:new_ending_index[1],
372                new_origin_index[0]:new_ending_index[0]
373            ]    
374            new_obj.topleft = np.array([
375                self.axis[2][new_origin_index[0]], # x
376                self.axis[1][new_origin_index[1]], # y
377                self.axis[0][new_origin_index[2]] # z
378            ])
379            new_obj.material_dict = self.material_dict     
380            new_obj.num_voxels = np.flip(new_obj.material_matrix.shape, 0)
381            new_obj.voxel_size = self.voxel_size   
382            new_obj.axis = self.calculateAxis()
383            new_obj.num_materials = self.num_materials
384            return new_obj

386    def crop_by_body_contour(self, body_index_range:Optional[np.ndarray] = None, 
387                             body_mask_shape:Optional[np.ndarray] = None,
388                             pth_dir_dicom:Optional[str]=None, 
389                             ):
390        r"""
391        Purpose: 
392            based on the given dicom structure file, crop the BrachyEgsphant object such 
393                that it only has the body contour. 
394        Inputs: 
395            - body_index_range:np.array :=  a 3 x 2 array holding the min and max on x, y and axis
396                [[x_min, x_max], [y_min, y_max], [z_min, z_max]]. If this is not available, provide
397                the third input. 
398        
399            - original_mask_dimensions:np.array := 1 x 3 array holding the dimension of the original mask. 
400                If this is not available, provide the third input. 
401
402            - pth_dir_dicom := pth_dir_dicom := path to the directory with the dicom files of a patient. 
403                it should contain both images and RTSTRUCT file. this input is used when the first 2 inputs 
404                are not available
405            
406        Outputs:
407            - Void := will crop out the material and density maps of self to have the range of the body contour 
408                    in the dicom structure file. It will also update the num_voxels, topleft and axis. only voxel_size will not change
409        """
410        
411        if body_index_range is None or body_mask_shape is None:
412            assert pth_dir_dicom is not None, "Either path to a dicom directory with dicom structure \
413                file should be given or body_index_range and body_mask_shape"
414            # body_index_range, body_mask_shape = get_structure_index_range(pth_dir_dicom)
415        # the body mask may have a different size than the material map, we normalize range to the dimension 
416        # of original mask and scale it to the dimension of the material map to get the body index range on the material image.  
417        scaled_body_index_range = (body_index_range / np.expand_dims(body_mask_shape, axis=1) * np.expand_dims(self.num_voxels, axis=1)).astype(int)
418        
419        self.crop_by_index(scaled_body_index_range, True)

422def _to_single_string(matrix:np.ndarray, deliminator:Optional[str]=""):
423    r"""
424    Purpose:
425        given a 3D matrix with string entries, this function concatenates all the
426            entries into a single string to be written to the file. 
427            "\n" is added at the end of each row and 
428    Input:
429        matrix := 3D ndarray full of string enteries
430        deliminator := the string text inbetween the enteries. 
431    Output:
432        a single string containing all the enteries with added \n at the end of each row of 
433            matrix and an addiation \n added to each slide in the input matrix
434     
435    """
436    matrix_single_string = []
437    for slide in matrix:
438        slide_single_string = []
439        for row in slide:
440            slide_single_string.append(
441                deliminator.join(row) + "\n"
442            )
443        matrix_single_string.append(deliminator.join(slide_single_string)+"\n")
444        
445    return "".join(matrix_single_string)

324        r"""
325        Purpose: 
326            given a range of indicies (mix and max on each axis), this function will crop

329            - self: BrachyEgsphant object

333            - Void := will crop out the material and density maps of self to have the range of the index range.

336            - None
337        """

347                new_origin_index[2]:new_ending_index[2], # z
348                new_origin_index[1]:new_ending_index[1], # y
349                new_origin_index[0]:new_ending_index[0] # x

355            ]

357                self.axis[2][new_origin_index[0]], # x
358                self.axis[1][new_origin_index[1]], # y
359                self.axis[0][new_origin_index[2]] # z
