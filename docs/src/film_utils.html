 73    def perform_fit(self):
 74        """
 75        Perform curve fitting to obtain optimal parameters and covariance matrices for red, green, and blue channels.
 76
 77        Returns:
 78        - r_opt: array_like
 79        Optimal parameters for red channel.
 80        - rp_cov: 2-D array
 81        Covariance matrix for the estimated parameters for red channel.
 82        - g_opt: array_like
 83        Optimal parameters for green channel.
 84        - gp_cov: 2-D array
 85        Covariance matrix for the estimated parameters for green channel.
 86        - b_opt: array_like
 87        Optimal parameters for blue channel.
 88        - bp_cov: 2-D array
 89        Covariance matrix for the estimated parameters for blue channel.
 90        """
 91        self.r_opt, self.rp_cov = curve_fit(self.dose_to_pv, self.doses, self.r_pv_fit, sigma=self.r_std)
 92        self.g_opt, self.gp_cov = curve_fit(self.dose_to_pv, self.doses, self.g_pv_fit, sigma=self.g_std)
 93        self.b_opt, self.bp_cov = curve_fit(self.dose_to_pv, self.doses, self.b_pv_fit, sigma=self.b_std)

455        r"""
456        Calculates the PV value for a given dose using the Lewis model.
457
458        Args:
459        d (float): The dose value.

470    def lewis_pv_to_dose(pv, a, b, c):
471        r"""
472        Converts a given PV value to a dose value using the Lewis formula.
473
474        Args:
475        pv (float): The PV value to convert to dose.
476        a (float): The 'a' parameter of the Lewis formula.
477        b (float): The 'b' parameter of the Lewis formula.
478        c (float): The 'c' parameter of the Lewis formula.
479
480        Returns:
481        float: The dose value corresponding to the given PV value.
482        """
483        return -c + b/(pv - a)

488        Converts a pixel value (PV) to a dose value using the Devic formula.

491        pv (float): Pixel value
492        a (float): Calibration coefficient
493        b (float): Calibration coefficient

495        Returns:
496        float: Dose value
497        """

503        Converts a measured dose value to a pixel value using the Devic formula.

508        b (float): The b parameter of the Devic formula.

511        float: The pixel value corresponding to the measured dose value.
